		end

				NS = SEL_COL;
			else
				NS = WAIT0;
		end
		SEL_COL: NS = KEY_SEL;
		WAIT1:
		begin
			if(KEY[2] == 1'b1)
				NS = SEL_LOC;
			else
				NS = WAIT1;
		end
		SEL_LOC: NS = KEY_SEL;
	endcase
end
